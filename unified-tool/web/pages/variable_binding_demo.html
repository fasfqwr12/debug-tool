💻 实现原理

1. 预设配置 (presets.json)

"boot_init": { "components": { "Line4_Unit": { "bind": "unit" }, "Line3_Unit": { "bind": "unit" }, // 绑定到 unit 变量，而不是固定值 } }

2. 组件定义 (components.json)

"Line4_Unit": { "type": "ModeSelector", "modes": { "m": ["Line4_Unit_U2"], "ft": ["Line4_Unit_U0"], "in": ["Line4_Unit_U1"] } }

3. 运行时映射 (lcd_runtime.js)

mapValueToMode(comp, varName, value) { if (varName === 'unit') { // 特殊映射: 整数 → 模式名 const unitMap = { 0: 'm', 1: 'ft', 2: 'in' }; return unitMap[value]; } }

4. 调用方式

// 状态机执行入口动作时 const variables = { unit: 1, ... }; LcdRuntime.applyPreset('boot_init', variables); // 内部处理: // 1. 发现 Line4_Unit 有 bind: "unit" // 2. 从 variables 读取 unit = 1 // 3. 映射 1 → "ft" // 4. 调用 setMode("Line4_Unit", "ft")
